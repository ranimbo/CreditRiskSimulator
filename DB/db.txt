-- ================================================
-- TABLE UTILISATEUR (abstraite)
-- ================================================
CREATE TABLE utilisateur (
    id      INT AUTO_INCREMENT PRIMARY KEY,
    nom     VARCHAR(100),
    email   VARCHAR(150) UNIQUE,
    mot_de_passe VARCHAR(255)
);

-- ================================================
-- ADMIN
-- ================================================
CREATE TABLE admin (
    id                  INT PRIMARY KEY,
    matricule           VARCHAR(50),
    role                VARCHAR(50),
    derniere_connexion  DATE,
    FOREIGN KEY (id) REFERENCES utilisateur(id) ON DELETE CASCADE
);

-- ================================================
-- AGENT BANCAIRE
-- ================================================
CREATE TABLE agent_bancaire (
    id            INT PRIMARY KEY,
    matricule     VARCHAR(50),
    agence        VARCHAR(100),
    date_creation DATE,
    FOREIGN KEY (id) REFERENCES utilisateur(id) ON DELETE CASCADE
);

-- ================================================
-- CLIENT
-- ================================================
CREATE TABLE client (
    id                   INT AUTO_INCREMENT PRIMARY KEY,
    nom                  VARCHAR(100),
    date_naissance       DATE,
    cin                  VARCHAR(50),
    situation_pro        ENUM('CDI','FONCTIONNAIRE','INDEPENDANT','SANS_EMPLOI'),
    revenu_mensuel_net   FLOAT,
    charges_mensuelles   FLOAT,
    anciennete_emploi    INT,
    historique_credit    TINYINT(1)
);

-- ================================================
-- DEMANDE CREDIT
-- ================================================
CREATE TABLE demande_credit (
    id                   INT AUTO_INCREMENT PRIMARY KEY,
    date_creation        DATETIME,
    montant_demande      FLOAT,
    duree                INT,
    type_credit          ENUM('CONSOMMATION','IMMOBILIER','PROFESSIONNEL'),
    taux_endettement_calc FLOAT,
    statut               VARCHAR(50),
    agent_id             INT,
    client_id            INT,
    FOREIGN KEY (agent_id)  REFERENCES agent_bancaire(id),
    FOREIGN KEY (client_id) REFERENCES client(id)
);

-- ================================================
-- CRITERE SCORING
-- ================================================
CREATE TABLE critere_scoring (
    id        INT AUTO_INCREMENT PRIMARY KEY,
    libelle   VARCHAR(100),
    poids     FLOAT,
    points_max INT,
    actif     TINYINT(1),
    ordre     INT
);

-- ================================================
-- SCORE
-- ================================================
CREATE TABLE score (
    id                 INT AUTO_INCREMENT PRIMARY KEY,
    valeur_totale      INT,
    date_calcul        DATETIME,
    detail_par_critere JSON,
    version_moteur     VARCHAR(50),
    temps_calcul       FLOAT,
    demande_id         INT,
    FOREIGN KEY (demande_id) REFERENCES demande_credit(id)
);

-- ================================================
-- DECISION
-- ================================================
CREATE TABLE decision (
    id                    INT AUTO_INCREMENT PRIMARY KEY,
    resultat              ENUM('ACCORDE','A_ANALYSER','REFUSE'),
    justification         TEXT,
    facteurs_favorables   JSON,
    facteurs_defavorables JSON,
    date_decision         DATETIME,
    score_id              INT,
    FOREIGN KEY (score_id) REFERENCES score(id)
);

-- ================================================
-- RAPPORT SIMULATION
-- ================================================
CREATE TABLE rapport_simulation (
    id               INT AUTO_INCREMENT PRIMARY KEY,
    date_generation  DATETIME,
    format           ENUM('PDF','EXCEL','CSV'),
    contenu          LONGBLOB,
    taille           INT,
    score_id         INT,
    decision_id      INT,
    FOREIGN KEY (score_id)    REFERENCES score(id),
    FOREIGN KEY (decision_id) REFERENCES decision(id)
);
